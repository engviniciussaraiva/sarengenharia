end if; if p_id is null then insert into public.sar_site_paginas ( slug, nome, titulo_seo, descricao_seo, arquivo, ordem, ativo ) values ( v_slug, trim(p_nome), nullif(trim(coalesce(p_titulo_seo, '')), ''), nullif(trim(coalesce(p_descricao_seo, '')), ''), coalesce( nullif(trim(coalesce(p_arquivo, '')), ''), v_slug || '.html' ), coalesce(p_ordem, 0), coalesce(p_ativo, true) ) returning id into v_id; else update public.sar_site_paginas set slug = v_slug, nome = trim(p_nome), titulo_seo = nullif(trim(coalesce(p_titulo_seo, '')), ''), descricao_seo = nullif(trim(coalesce(p_descricao_seo, '')), ''), arquivo = coalesce( nullif(trim(coalesce(p_arquivo, '')), ''), v_slug || '.html' ), ordem = coalesce(p_ordem, 0), ativo = coalesce(p_ativo, true) where id = p_id returning id into v_id;
